【範例. Chick Weight 家禽資料集－Stage C-2 更新版】

# ============================================================
# Stage 0：Kaggle Source 0~21 Pre-Train
# 目的：先讓模型學習 Kaggle 0~21 日齡雞隻早期生長趨勢
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode pre-train \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# Stage A Baseline：FwuSow Target 0~21 Without Transfer Learning
# 目的：建立福壽 0~21 日齡 Target baseline
# Target：dataset/target/FwuSow_Poultry_0~21
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode without-transfer-learning \
  --target-name "FwuSow_Poultry_0~21" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# Stage A：Kaggle Source 0~21 -> FwuSow Target 0~21
# 實驗定位：主實驗 TL，驗證 0~21 日齡對齊條件下 TL 是否有效
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# Target：dataset/target/FwuSow_Poultry_0~21
# 模型設定：Partial Fine-tuning
# 目的：產生 Stage A 權重，作為 Stage B 分階段遷移之 pre-model
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --target-name "FwuSow_Poultry_0~21" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# Stage A Freeze：Kaggle Source 0~21 -> FwuSow Target 0~21
# 實驗定位：Stage A 的 Freeze 對照組
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# Target：dataset/target/FwuSow_Poultry_0~21
# 模型設定：Freeze Transfer Learning
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --freeze \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --target-name "FwuSow_Poultry_0~21" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# 建議：將 Stage A 產生的最佳權重另存為穩定路徑
# 例如：
# reports/Chick Weight Result/pre-train/StageA_Kaggle0_21_to_FwuSow0_21/best_model.hdf5
# 目的：避免 Stage B 使用錯誤權重
# 注意：若沒有手動複製，Stage B 的 --pre-model-path 必須改成實際 Stage A 輸出路徑
# ============================================================


# ============================================================
# Stage B Baseline：FwuSow Target 0~31 Without Transfer Learning
# 實驗定位：分階段遷移的完整日齡 baseline
# Target：dataset/target/FwuSow_Poultry_0_31_reuse0_21_scaler
# 注意：此 baseline 使用 reuse0_21_scaler，可與 Stage C-1 共用；
#       不可直接與 Stage C-2 full scaler baseline 混用。
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode without-transfer-learning \
  --target-name "FwuSow_Poultry_0_31_reuse0_21_scaler" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# Stage B：Stage A 權重 -> FwuSow Target 0~31
# 實驗定位：分階段遷移
# Source/Pre-model：Stage A, Kaggle 0~21 -> FwuSow 0~21
# Target：dataset/target/FwuSow_Poultry_0_31_reuse0_21_scaler
# 模型設定：Partial Fine-tuning
# 目的：檢查 Stage A 權重是否能改善 0~31 後期延伸任務
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --target-name "FwuSow_Poultry_0_31_reuse0_21_scaler" \
  --pre-model-path "reports/Chick Weight Result/pre-train/StageA_Kaggle0_21_to_FwuSow0_21/best_model.hdf5" \
  --source-name "StageA_Kaggle0_21_to_FwuSow0_21" \
  --seed 1234 \
  --nb-epochs 6500 \
  --period 3 \
  --learning-rate 1e-9


# ============================================================
# Stage B Freeze：Stage A 權重 -> FwuSow Target 0~31
# 實驗定位：分階段遷移 Freeze 對照組
# Source/Pre-model：Stage A, Kaggle 0~21 -> FwuSow 0~21
# Target：dataset/target/FwuSow_Poultry_0_31_reuse0_21_scaler
# 模型設定：Freeze Transfer Learning
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --freeze \
  --target-name "FwuSow_Poultry_0_31_reuse0_21_scaler" \
  --pre-model-path "reports/Chick Weight Result/pre-train/StageA_Kaggle0_21_to_FwuSow0_21/best_model.hdf5" \
  --source-name "StageA_Kaggle0_21_to_FwuSow0_21" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3 \
  --learning-rate 1e-9


# ============================================================
# Stage C-1 Baseline：FwuSow Target 0~31 Without Transfer Learning
# 實驗定位：直接遷移對照 baseline（reuse0_21_scaler 版本）
# Target：dataset/target/FwuSow_Poultry_0_31_reuse0_21_scaler
# 注意：若 Stage B Baseline 已執行過同一 target，這段可略過，共用 baseline 結果。
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode without-transfer-learning \
  --target-name "FwuSow_Poultry_0_31_reuse0_21_scaler" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# Stage C-1：Kaggle Source 0~21 Pre-Train -> FwuSow Target 0~31
# 實驗定位：Direct Transfer，直接遷移對照組（reuse0_21_scaler 版本）
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# Target：dataset/target/FwuSow_Poultry_0_31_reuse0_21_scaler
# Pre-model：Kaggle Source 0~21 pre-train best_model.hdf5
# 模型設定：Partial Fine-tuning
# 目的：檢查不經過 FwuSow 0~21 中介 fine-tuning 時，
#       Kaggle 權重直接遷移到 FwuSow 0~31 的效果。
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --target-name "FwuSow_Poultry_0_31_reuse0_21_scaler" \
  --pre-model-path "reports/Chick Weight Result/pre-train/Kaggle Weight vs Age of Chicks/best_model.hdf5" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3 \
  --learning-rate 1e-9


# ============================================================
# Stage C-1 Freeze：Kaggle Source 0~21 Pre-Train -> FwuSow Target 0~31
# 實驗定位：Direct Transfer Freeze 對照組（reuse0_21_scaler 版本）
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# Target：dataset/target/FwuSow_Poultry_0_31_reuse0_21_scaler
# Pre-model：Kaggle Source 0~21 pre-train best_model.hdf5
# 模型設定：Freeze Transfer Learning
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --freeze \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --target-name "FwuSow_Poultry_0_31_reuse0_21_scaler" \
  --pre-model-path "reports/Chick Weight Result/pre-train/Kaggle Weight vs Age of Chicks/best_model.hdf5" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3 \
  --learning-rate 1e-9


# ============================================================
# Stage C-2 Baseline：FwuSow Target 0~31 Without Transfer Learning
# 實驗定位：Stage C-2 full scaler 版本之 Target baseline
# Target：dataset/target/FwuSow_Poultry_0_31_fit0_31_full_scaler
# Scaler：FwuSow 0~31 全區間 fit MinMaxScaler 後再切 train/test
# 注意：此版本屬於 full scaler 補充實驗；正式嚴謹版本仍建議另做 train-only scaler。
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode without-transfer-learning \
  --target-name "FwuSow_Poultry_0_31_fit0_31_full_scaler" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3


# ============================================================
# Stage C-2：Kaggle Source 0~21 Pre-Train -> FwuSow Target 0~31
# 實驗定位：Direct Transfer 補充實驗（fit0_31_full_scaler 版本）
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# Target：dataset/target/FwuSow_Poultry_0_31_fit0_31_full_scaler
# Pre-model：Kaggle Source 0~21 pre-train best_model.hdf5
# 模型設定：Partial Fine-tuning
# 目的：檢查在 FwuSow 0~31 全區間正規化尺度下，
#       Kaggle 0~21 預訓練權重是否能改善完整 Target task 預測。
# ============================================================

python main.py --out-dir "Chick Weight Result" \
  --train-mode transfer-learning \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --target-name "FwuSow_Poultry_0_31_fit0_31_full_scaler" \
  --pre-model-path "reports/Chick Weight Result/pre-train/Kaggle Weight vs Age of Chicks/best_model.hdf5" \
  --seed 1234 \
  --nb-epochs 5000 \
  --period 3 \
  --learning-rate 1e-5


# ============================================================
# Stage C-2 Freeze：Kaggle Source 0~21 Pre-Train -> FwuSow Target 0~31
# 實驗定位：Direct Transfer Freeze 補充實驗（fit0_31_full_scaler 版本）
# Source：dataset/source/Kaggle Weight vs Age of Chicks
# Target：dataset/target/FwuSow_Poultry_0_31_fit0_31_full_scaler
# Pre-model：Kaggle Source 0~21 pre-train best_model.hdf5
# 模型設定：Freeze Transfer Learning
# 目的：比較 fit0_31_full_scaler 設定下 Freeze 與 Partial Fine-tuning 的差異。
# ============================================================

  --train-mode transfer-learning \
  --freeze \
  --source-name "Kaggle Weight vs Age of Chicks" \
  --target-name "FwuSow_Poultry_0_31_fit0_31_full_scaler" \
  --pre-model-path "reports/Chick Weight Result/pre-train/Kaggle Weight vs Age of Chicks/best_model.hdf5" \

  --period 3 \
